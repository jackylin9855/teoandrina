  <script>
    /*
      👇【照片在程式碼裡放置的位置】👇

    */
    const SAMPLE_RECORDS = [
      {
        id: 'rec-1',
        subject: 'Teo',
        gen: '二代鼠',
        date: '2025/02/13',
        weight: '58',
        tail: '圓潤充實 ',
        notes: 'Teo 二代鼠，以前顏值很高又很愛乾淨，現在有變好一點了。',
        image: './S__28885000.jpg'
      },
      {

        subject: 'Andrina',
        gen: '一代鼠',
        date: '2024/02/13-2025/09/18',
        weight: '65',

        notes: 'Andrina 始祖鼠，從魚中魚買來的，很愛趴在尿上腹部永遠黃黃的',
        image: './S__28885001.jpg'
      }

    let records = [];
    let currentFilter = 'all';
    let currentSelectedImageBase64 = null;

    window.addEventListener('DOMContentLoaded', () => {
      loadRecords();
      const dateInput = document.getElementById('input-date');
      if (dateInput) dateInput.valueAsDate = new Date();
    });

    function loadRecords() {
      try {
        // 升級版本號至 v8，強制讓修改後的 SAMPLE_RECORDS 生效
        const saved = localStorage.getItem('gerbil_records_v8');
        if (saved && JSON.parse(saved).length > 0) {
          records = JSON.parse(saved);
        } else {
          records = [...SAMPLE_RECORDS];
          saveToStorage();
        }
      } catch (err) {
        console.error("Storage Load Error:", err);
        records = [...SAMPLE_RECORDS];
      }
      renderGallery();
      updateStats();
    }

    function saveToStorage() {
      try {
        localStorage.setItem('gerbil_records_v8', JSON.stringify(records));
      } catch (e) {
        showToast("⚠️ 容量已滿，請刪除部分舊照或備份！", "error");
      }
    }

    function renderGallery() {
      const grid = document.getElementById('gallery-grid');
      const emptyState = document.getElementById('empty-state');
      if (!grid || !emptyState) return;

      grid.innerHTML = '';
      const filtered = records.filter(r => currentFilter === 'all' || r.subject === currentFilter);

      if (filtered.length === 0) {
        emptyState.classList.remove('hidden');
      } else {
        emptyState.classList.add('hidden');
        
        filtered.forEach(record => {
          const card = document.createElement('div');
          card.className = "bg-white rounded-2xl overflow-hidden shadow-sm hover:shadow-xl transition duration-300 border border-amber-100 group cursor-pointer flex flex-col";
          card.onclick = () => openLightbox(record.id);

          const subjectColor = record.subject === 'Teo' 
            ? 'bg-blue-50 text-blue-700 border border-blue-100' 
            : 'bg-indigo-50 text-indigo-700 border border-indigo-100';

          // 修正：Teo 為二代鼠，Andrina 為一代鼠
          const genLabel = record.gen || (record.subject === 'Teo' ? '二代鼠' : '一代鼠');

          card.innerHTML = `
            <div class="aspect-square w-full bg-slate-100 overflow-hidden relative">
              <img src="${record.image}" class="w-full h-full object-cover group-hover:scale-105 transition duration-500" alt="${record.subject}">
              <span class="absolute top-3 left-3 text-[11px] font-bold px-2.5 py-1 rounded-full backdrop-blur-md shadow-sm ${subjectColor}">
                ♂ ${record.subject} · ${genLabel}
              </span>
            </div>
            <div class="p-4 flex-1 flex flex-col justify-between space-y-2">
              <div>
                <div class="flex justify-between items-center text-xs text-slate-400 mb-1">
                  <span><i class="fa-regular fa-calendar mr-1"></i>${record.date}</span>
                  ${record.weight ? `<span class="font-bold text-amber-700"><i class="fa-solid fa-weight-scale mr-1"></i>${record.weight}g</span>` : ''}
                </div>
                <p class="text-xs text-slate-600 line-clamp-2 leading-relaxed">${record.notes || '沒有備註日記...'}</p>
              </div>
              <div class="pt-2 border-t border-slate-50 flex justify-between items-center text-[11px] text-slate-400">
                <span>尾巴: ${record.tail ? record.tail.split(' ')[0] : '正常'}</span>
                <span class="text-amber-600 font-medium group-hover:underline">看大圖 →</span>
              </div>
            </div>
          `;
          grid.appendChild(card);
        });
      }
    }

    function updateStats() {
      const totalElem = document.getElementById('stat-total-count');
      const teoElem = document.getElementById('stat-teo-latest');
      const andrinaElem = document.getElementById('stat-andrina-latest');

      if (totalElem) totalElem.textContent = `${records.length} 篇`;

      const teoRecords = records.filter(r => r.subject === 'Teo' && r.weight).sort((a,b) => new Date(b.date) - new Date(a.date));
      const andrinaRecords = records.filter(r => r.subject === 'Andrina' && r.weight).sort((a,b) => new Date(b.date) - new Date(a.date));

      if (teoElem) teoElem.textContent = teoRecords.length > 0 ? `最新: ${teoRecords[0].weight} g` : `最新: -- g`;
      if (andrinaElem) andrinaElem.textContent = andrinaRecords.length > 0 ? `最新: ${andrinaRecords[0].weight} g` : `最新: -- g`;
    }

    function handleFileSelect(e) {
      const file = e.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(event) {
        const img = new Image();
        img.onload = function() {
          const canvas = document.createElement('canvas');
          let width = img.width;
          let height = img.height;
          const maxDim = 800;

          if (width > height) {
            if (width > maxDim) {
              height = Math.round((height * maxDim) / width);
              width = maxDim;
            }
          } else {
            if (height > maxDim) {
              width = Math.round((width * maxDim) / height);
              height = maxDim;
            }
          }

          canvas.width = width;
          canvas.height = height;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0, width, height);

          currentSelectedImageBase64 = canvas.toDataURL('image/jpeg', 0.75);

          const previewImg = document.getElementById('preview-img');
          const uploadPrompt = document.getElementById('upload-prompt');
          if (previewImg && uploadPrompt) {
            previewImg.src = currentSelectedImageBase64;
            previewImg.classList.remove('hidden');
            uploadPrompt.classList.add('hidden');
          }
        };
        img.src = event.target.result;
      };
      reader.readAsDataURL(file);
    }

    function handleSaveRecord(e) {
      e.preventDefault();

      if (!currentSelectedImageBase64) {
        showToast("請先選擇一張照片喔！", "error");
        return;
      }

      const subjectRadio = document.querySelector('input[name="subject"]:checked');
      const subject = subjectRadio ? subjectRadio.value : 'Teo';
      // 修正：儲存新紀錄時的預設標籤
      const gen = subject === 'Teo' ? '二代鼠' : '一代鼠';
      const date = document.getElementById('input-date').value;
      const weight = document.getElementById('input-weight').value;
      const tail = document.getElementById('input-tail').value;
      const notes = document.getElementById('input-notes').value;

      const newRecord = {
        id: 'rec-' + Date.now(),
        subject,
        gen,
        date,
        weight,
        tail,
        notes,
        image: currentSelectedImageBase64
      };

      records.unshift(newRecord);
      saveToStorage();
      renderGallery();
      updateStats();
      closeAddModal();
      showToast("成功新增照片！🐾");
    }

    function openLightbox(id) {
      const record = records.find(r => r.id === id);
      if (!record) return;

      document.getElementById('lb-img').src = record.image;
      document.getElementById('lb-date').textContent = record.date;
      document.getElementById('lb-weight').textContent = record.weight ? `${record.weight} g` : '未紀錄';
      document.getElementById('lb-tail').textContent = record.tail || '正常';
      document.getElementById('lb-notes').textContent = record.notes || '沒有寫備註喔～';

      const badge = document.getElementById('lb-subject-badge');
      // 修正：大圖視窗中的標籤顯示
      const genText = record.gen || (record.subject === 'Teo' ? '二代鼠' : '一代鼠');
      if (badge) {
        badge.textContent = `♂ ${record.subject} (${genText})`;
        badge.className = `px-3 py-1 rounded-full text-xs font-bold ${
          record.subject === 'Teo' ? 'bg-blue-100 text-blue-700' : 'bg-indigo-100 text-indigo-700'
        }`;
      }

      document.getElementById('lb-delete-btn').onclick = () => deleteRecord(record.id);
      document.getElementById('modal-lightbox').classList.remove('hidden');
    }

    function closeLightbox() {
      document.getElementById('modal-lightbox').classList.add('hidden');
    }

    function deleteRecord(id) {
      records = records.filter(r => r.id !== id);
      saveToStorage();
      renderGallery();
      updateStats();
      closeLightbox();
      showToast("已刪除該篇照片紀錄");
    }

    function openAddModal() {
      const form = document.getElementById('record-form');
      if (form) form.reset();
      const dateInput = document.getElementById('input-date');
      if (dateInput) dateInput.valueAsDate = new Date();
      currentSelectedImageBase64 = null;

      const previewImg = document.getElementById('preview-img');
      const uploadPrompt = document.getElementById('upload-prompt');
      if (previewImg) previewImg.classList.add('hidden');
      if (uploadPrompt) uploadPrompt.classList.remove('hidden');

      document.getElementById('modal-add').classList.remove('hidden');
    }

    function closeAddModal() {
      document.getElementById('modal-add').classList.add('hidden');
    }

    function setFilter(cat) {
      currentFilter = cat;
      document.querySelectorAll('.filter-btn').forEach(btn => {
        btn.classList.remove('bg-amber-600', 'text-white');
        btn.classList.add('bg-white', 'text-slate-600');
      });
      const activeBtn = document.getElementById(`filter-${cat}`);
      if (activeBtn) {
        activeBtn.classList.remove('bg-white', 'text-slate-600');
        activeBtn.classList.add('bg-amber-600', 'text-white');
      }
      renderGallery();
    }

    function showToast(msg, type = "success") {
      const toast = document.getElementById('toast');
      const toastMsg = document.getElementById('toast-msg');
      const toastIcon = document.getElementById('toast-icon');

      if (!toast || !toastMsg || !toastIcon) return;

      toastMsg.textContent = msg;
      toastIcon.className = type === "error" 
        ? "fa-solid fa-circle-exclamation text-red-400 text-base" 
        : "fa-solid fa-circle-check text-emerald-400 text-base";

      toast.classList.remove('translate-y-20', 'opacity-0');
      setTimeout(() => {
        toast.classList.add('translate-y-20', 'opacity-0');
      }, 3000);
    }
